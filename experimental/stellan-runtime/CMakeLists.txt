cmake_minimum_required(VERSION 3.20)
project(stellan_runtime VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
add_library(stellan-core src/runtime.cpp src/simd.cpp src/quant.cpp src/executor.cpp src/tokenizer.cpp src/lite.cpp src/max.cpp src/kv_cache.cpp src/sha256.cpp src/sha256_x86.cpp src/session.cpp src/host_input.cpp src/vision.cpp)
target_link_libraries(stellan-core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(stellan-core PRIVATE Psapi)
endif()
target_include_directories(stellan-core PUBLIC src)
if(MSVC)
  target_compile_options(stellan-core PUBLIC /utf-8 PRIVATE /W4 /WX)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|x86_64")
    set_source_files_properties(src/simd.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX2)
  endif()
else()
  target_compile_options(stellan-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(stellan-runtime src/main.cpp)
set_target_properties(stellan-runtime PROPERTIES OUTPUT_NAME stellan-run)
target_link_libraries(stellan-runtime PRIVATE stellan-core)
# Mbed TLS (Apache-2.0) gives Host HTTPS for GitHub sign-in, updates and activation. Pinned to the v3.6.8 commit
# and built from source; only the libraries, no programs or tests.
enable_language(C)
include(FetchContent)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git GIT_TAG eea50bd0fe804a43caff1ff97318331a5dad28c0 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(mbedtls)
# Monocypher (BSD-2-Clause or CC0) for Ed25519 signature checks (activation certificates, update feeds); pinned to the
# 4.0.2 commit and built from source. Only the RFC 8032 Ed25519 verifier is used.
FetchContent_Declare(monocypher GIT_REPOSITORY https://github.com/LoupVaillant/Monocypher.git GIT_TAG 0d85f98c9d9b0227e42cf795cb527dff372b40a4 GIT_SHALLOW FALSE)
FetchContent_GetProperties(monocypher)
if(NOT monocypher_POPULATED)
  FetchContent_Populate(monocypher)
endif()
add_library(monocypher STATIC ${monocypher_SOURCE_DIR}/src/monocypher.c ${monocypher_SOURCE_DIR}/src/optional/monocypher-ed25519.c)
target_include_directories(monocypher PUBLIC ${monocypher_SOURCE_DIR}/src ${monocypher_SOURCE_DIR}/src/optional)
add_executable(stellan-host src/host.cpp src/workspace.cpp src/mcp_bridge.cpp src/front_door.cpp src/json.cpp src/protocol.cpp src/protocol_check.cpp src/agent_core.cpp src/engine_client.cpp src/https.cpp src/mcp_wire.cpp src/mcp_client.cpp src/crypto.cpp src/inflate.cpp src/model_server.cpp src/updater.cpp src/documents.cpp src/pdf.cpp src/skills.cpp src/hooks.cpp src/process.cpp src/gitignore.cpp)
target_link_libraries(stellan-host PRIVATE stellan-core mbedtls mbedx509 mbedcrypto monocypher)
if(WIN32)
  target_link_libraries(stellan-host PRIVATE ws2_32)
endif()
# The updater with a test signing key and local feeds allowed, for tests only: never packaged (scripts copy stellan-host by name).
add_executable(stellan-update-testing src/updater_testing.cpp src/updater.cpp src/json.cpp src/https.cpp src/crypto.cpp src/inflate.cpp)
target_compile_definitions(stellan-update-testing PRIVATE STELLAN_UPDATE_TESTING)
target_link_libraries(stellan-update-testing PRIVATE stellan-core mbedtls mbedx509 mbedcrypto monocypher)
if(WIN32)
  target_link_libraries(stellan-update-testing PRIVATE ws2_32)
endif()
add_executable(stellan-runtime-tests tests/core.cpp)
target_link_libraries(stellan-runtime-tests PRIVATE stellan-core)
add_executable(stellan-cache-bench EXCLUDE_FROM_ALL bench/cache-growth.cpp)
target_link_libraries(stellan-cache-bench PRIVATE stellan-core)
add_executable(stellan-batch-bench EXCLUDE_FROM_ALL bench/quantized-batch.cpp)
target_link_libraries(stellan-batch-bench PRIVATE stellan-core)
enable_testing()
add_test(NAME core COMMAND stellan-runtime-tests)
add_test(NAME core-scalar COMMAND stellan-runtime-tests)
add_test(NAME core-avx2 COMMAND stellan-runtime-tests)
set_tests_properties(core-avx2 PROPERTIES ENVIRONMENT "STELLAN_RUNTIME_NO_VNNI=1")
set_tests_properties(core-scalar PROPERTIES ENVIRONMENT "STELLAN_RUNTIME_SCALAR=1")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
  add_test(NAME core-arm-untiled COMMAND stellan-runtime-tests)
  set_tests_properties(core-arm-untiled PROPERTIES ENVIRONMENT "STELLAN_RUNTIME_ARM_UNTILED=1")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME gguf COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gguf.py $<TARGET_FILE:stellan-runtime>)
add_test(NAME splits COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/splits.py $<TARGET_FILE:stellan-runtime>)
add_test(NAME models COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/models.py $<TARGET_FILE:stellan-runtime>)
add_test(NAME vision COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/vision.py $<TARGET_FILE:stellan-runtime>)
install(TARGETS stellan-runtime stellan-host RUNTIME DESTINATION bin)

find_program(STELLAN_NODE node)
if(STELLAN_NODE)
  add_test(NAME session-bridge COMMAND ${CMAKE_COMMAND} -E env "STELLAN_TEST_ENGINE=$<TARGET_FILE:stellan-runtime>" "STELLAN_PYTHON=${Python3_EXECUTABLE}" ${STELLAN_NODE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/session.test.mjs)
endif()
add_test(NAME native-host COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/host.py $<TARGET_FILE:stellan-runtime> $<TARGET_FILE:stellan-host>)

add_test(NAME native-workspace COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/workspace.py $<TARGET_FILE:stellan-host>)

if(STELLAN_NODE)
  add_test(NAME native-workspace-client COMMAND ${CMAKE_COMMAND} -E env "STELLAN_TEST_HOST=$<TARGET_FILE:stellan-host>" ${STELLAN_NODE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/workspace.test.mjs)
  add_test(NAME native-mcp-bridge COMMAND ${CMAKE_COMMAND} -E env "STELLAN_TEST_HOST=$<TARGET_FILE:stellan-host>" ${STELLAN_NODE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp-bridge.test.mjs)
  add_test(NAME native-state-client COMMAND ${CMAKE_COMMAND} -E env "STELLAN_TEST_HOST=$<TARGET_FILE:stellan-host>" ${STELLAN_NODE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/state.test.mjs)
endif()

add_test(NAME native-state-store COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/state.py $<TARGET_FILE:stellan-host>)
