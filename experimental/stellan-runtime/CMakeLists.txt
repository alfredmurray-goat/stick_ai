cmake_minimum_required(VERSION 3.20)
project(stellan_runtime VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
add_library(stellan-core src/runtime.cpp src/simd.cpp src/quant.cpp src/executor.cpp src/tokenizer.cpp src/lite.cpp src/max.cpp src/kv_cache.cpp src/sha256.cpp src/sha256_x86.cpp src/session.cpp src/host_input.cpp)
target_link_libraries(stellan-core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(stellan-core PRIVATE Psapi)
endif()
target_include_directories(stellan-core PUBLIC src)
if(MSVC)
  target_compile_options(stellan-core PUBLIC /utf-8 PRIVATE /W4 /WX)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|x86_64")
    set_source_files_properties(src/simd.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX2)
  endif()
else()
  target_compile_options(stellan-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(stellan-runtime src/main.cpp)
set_target_properties(stellan-runtime PROPERTIES OUTPUT_NAME stellan-run)
target_link_libraries(stellan-runtime PRIVATE stellan-core)
add_executable(stellan-host src/host.cpp)
target_link_libraries(stellan-host PRIVATE stellan-core)
add_executable(stellan-runtime-tests tests/core.cpp)
target_link_libraries(stellan-runtime-tests PRIVATE stellan-core)
add_executable(stellan-cache-bench EXCLUDE_FROM_ALL bench/cache-growth.cpp)
target_link_libraries(stellan-cache-bench PRIVATE stellan-core)
add_executable(stellan-batch-bench EXCLUDE_FROM_ALL bench/quantized-batch.cpp)
target_link_libraries(stellan-batch-bench PRIVATE stellan-core)
enable_testing()
add_test(NAME core COMMAND stellan-runtime-tests)
add_test(NAME core-scalar COMMAND stellan-runtime-tests)
set_tests_properties(core-scalar PROPERTIES ENVIRONMENT "STELLAN_RUNTIME_SCALAR=1")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME gguf COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gguf.py $<TARGET_FILE:stellan-runtime>)
add_test(NAME splits COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/splits.py $<TARGET_FILE:stellan-runtime>)
add_test(NAME models COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/models.py $<TARGET_FILE:stellan-runtime>)
install(TARGETS stellan-runtime stellan-host RUNTIME DESTINATION bin)

find_program(STELLAN_NODE node)
if(STELLAN_NODE)
  add_test(NAME session-bridge COMMAND ${CMAKE_COMMAND} -E env "STELLAN_TEST_ENGINE=$<TARGET_FILE:stellan-runtime>" "STELLAN_PYTHON=${Python3_EXECUTABLE}" ${STELLAN_NODE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/session.test.mjs)
endif()
add_test(NAME native-host COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/host.py $<TARGET_FILE:stellan-runtime> $<TARGET_FILE:stellan-host>)
